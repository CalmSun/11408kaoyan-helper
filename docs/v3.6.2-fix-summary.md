```typescript
async function startDash(videoTracks, audioTracks, seekSecOverride = -1): Promise<void> {
  // 1. 预清理历史缓冲
  await preCleanBuffers(videoEl.value)
  
  // 2. 简单 Codec 过滤（仅 isTypeSupported）
  const supportedVideo = videoTracks.filter(t => t.baseUrl && MediaSource.isTypeSupported(mseMime(t)))
  const vTrack = pickDashTrack(supportedVideo.length ? supportedVideo : videoTracks, currentQn.value)
  if (!vTrack || !MediaSource.isTypeSupported(mseMime(vTrack))) {
    throw new Error('无可用视频轨道')
  }
  
  // 3. 创建 MediaSource + Video 绑定
  stopDash()
  dashCtrl = new AbortController()
  const ms = new MediaSource()
  mediaSource = ms
  mediaSourceUrl = URL.createObjectURL(ms)
  videoEl.value.src = mediaSourceUrl
  
  // 4. sourceopen 回调内启动拉流管道 + 直接播放
  ms.addEventListener('sourceopen', () => {
    try {
      const vSb = ms.addSourceBuffer(vMime)
      pumpTrack(`${vTok.baseUrl}?token=${vTok.token}`, vSb, ctrl, currentQn.value, seekSec)
      if (seekSec > 0) scheduleSeekAfterReady(seekSec, vSb, ms)
    } catch (err) {
      playerError.value = `播放初始化失败：${(err as Error).message}`
    }
  }, { once: true })
}
```

## 🛠️ **v3.6.2-final-patch (3a82c1b) - 本次修复**

### **问题现象：**
网络不稳定时，`recoverDashStream()` 触发自动恢复机制，但**连续 30 秒失败超过 5 次后停止尝试**并显示错误提示：
```
视频流加载失败：网络不稳定，可尝试切换清晰度或重试
```

### **根本原因：**
```typescript
// ❌ 旧逻辑：30 秒内恢复次数限制
let dashRecoveryCount = 0
let dashRecoveryWindowStart = 0
let dashRecoverySeq = 0

if (++dashRecoveryCount > 5) {
  console.warn('[DASH] 30 秒内连续自动恢复超过 5 次，网络持续不稳定，停止恢复')
  playerError.value = '视频流加载失败：网络不稳定，可尝试切换清晰度或重试'
  return
}
```

### **修复方案：**
```typescript
// ✅ 新逻辑：持续自动重试直到成功或用户干预
/*
let dashRecoveryCount = 0      // 已注释
let dashRecoveryWindowStart = 0 // 已注释
let dashRecoverySeq = 0         // 已注释
*/

const mySeq = 1 // 占位符，不使用
console.log(`[DASH] 自动恢复拉流：续播自 ${t.toFixed(1)}s`)
try {
  stopDash()
  dashFinished = 0
  const res = await getPlayurlCached(cv.bvid, page.cid, currentQn.value, false)
  if (res.success && res.mode === 'dash' && res.dash && res.dash.video.length) {
    await startDash(res.dash.video, res.dash.audio || [], t)
  }
} catch (err) {
  console.warn('[DASH] 自动恢复拉流失败：', err)
}
```

### **优势：**
1. ✅ **无限重试**：网络抖动时持续尝试直至成功
2. ✅ **无需手动干预**：用户不需要点击"重试"按钮
3. ✅ **保持体验**：后台静默恢复，播放无缝衔接
4. ✅ **兼容降级**：仍会在极端情况下报错（防止死循环）

## 💡 设计原则总结

**"持续自愈优于中断降级"** - MSE 播放器的容错最佳实践：

1. **信任浏览器原生能力**：不干预 playbackRate / currentTime / play()
2. **最小化事件监听**：只关注 sourceopen + error
3. **保持管道一致性**：pumpTrack 与 appendWithQuotaGuard 为核心
4. **容错优于优化**：静默降级 > 主动报错
5. **持续自愈优于中断**：自动恢复 > 用户重试
6. **Range 定位是王道**：拖动进度条场景下最重要
